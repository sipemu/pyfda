% inner padding, breakable long lines with an indented hooked continuation.
\lstdefinestyle{fdarscommon}{
  basicstyle=\ttfamily\footnotesize,
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
  breaklines=true,
  breakatwhitespace=false,
  breakindent=1.6em,
  framesep=7pt,
  framexleftmargin=7pt,
  xleftmargin=7pt,
  xrightmargin=7pt,
  framerule=0.5pt,
  postbreak=\mbox{\textcolor{codecom}{$\hookrightarrow$}\space},
}

% paper/paper.tex — minimal compiling skeleton for the fdars software paper
% Bibliography: natbib + BibTeX (plain BibTeX only; CI uses tectonic)
% The date field is empty intentionally for deterministic output.
\documentclass[12pt,a4paper]{article}

% Bibliography: natbib + BibTeX (plain BibTeX only)
\usepackage[numbers,sort&compress]{natbib}

% Basic packages for a software paper
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{graphicx}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{booktabs}
\usepackage{adjustbox}
\usepackage{listings}
% hyperref is loaded LATE (after the packages above) so it patches them correctly.
% colorlinks makes internal cross-references, citations, and URLs clearly clickable
% (the default bordered links are easy to miss). xurl loads AFTER hyperref and
% breaks long URLs/DOIs in the bibliography at arbitrary characters.
\PassOptionsToPackage{hyphens}{url}
\usepackage[colorlinks=true,linkcolor=blue,citecolor=blue,urlcolor=blue,linktoc=all]{hyperref}
\usepackage{xurl}

% Listing styles for executed capability-tour snippets (paper/snippets/*.tex).
% fdarsinput  — Python source block: boxed, monospace, keyword highlighting.
% fdarsoutput — Captured stdout block: plain monospace, no frame.
\lstdefinestyle{fdarsinput}{
  language=Python,
  basicstyle=\ttfamily\small,
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
  breaklines=true,
  breakatwhitespace=false,
  frame=single,
}
\lstdefinestyle{fdarsoutput}{
  basicstyle=\ttfamily\small,
  columns=fullflexible,
  keepspaces=true,
  breaklines=true,
  breakatwhitespace=false,
  frame=none,
}
% UTF-8 literate map: typeset Fdata repr's en-dash and multiplication sign
% cleanly inside listings (no special TeX escaping needed in the .tex source).
\lstset{
  literate=
    {--}{{--}}2
    {–}{{--}}1
    {×}{{$\times$}}1,
}
